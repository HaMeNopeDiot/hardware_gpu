# ============================================================
# RISC-V 32-bit (RV32IF) — Vertex Shader
# ============================================================
#
# Соглашение о вызовах / раскладка данных:
#   a0 = указатель на входные атрибуты (in_position vec4, in_color vec4, in_normal vec3)
#        смещения: 0..15  = in_position (4 float)
#                  16..31 = in_color    (4 float)
#                  32..43 = in_normal   (3 float)
#
#   a1 = указатель на UBO (uniform buffer object)
#        смещения:
#          0x00 = mvMatrix    row0 (vec4)  — %67
#          0x10 = mvMatrix    row1 (vec4)  — %69
#          0x20 = mvMatrix    row2 (vec4)  — %71
#          0x30 = mvMatrix    row3 (vec4)  — %73
#          0x40 = mvpMatrix   row0 (vec4)  — %11
#          0x50 = mvpMatrix   row1 (vec4)  — %13
#          0x60 = mvpMatrix   row2 (vec4)  — %15
#          0x70 = mvpMatrix   row3 (vec4)  — %17
#          0x80 = normalMatrix row0 (vec3) — %47
#          0x90 = normalMatrix row1 (vec3) — %49
#          0xa0 = normalMatrix row2 (vec3) — %51
#
#   a2 = указатель на выходные данные
#        смещения: 0..15  = gl_Position    (vec4)
#                  16..31 = vVaryingColor  (vec4)
#
# Регистры с плавающей точкой:
#   ft0-ft11 — временные
#   fa0-fa7  — аргументы / временные
#   fs0-fs11 — callee-saved (сохраняемые)
#
# Мы свободно используем ft0-ft11 и fs0-fs11 как рабочие регистры.
# ============================================================

    .text
    .globl  vertex_shader_main
    .type   vertex_shader_main, @function
    .align  2

vertex_shader_main:
    # -----------------------------------------------------------
    # Пролог: сохраняем callee-saved FP-регистры и ra
    # -----------------------------------------------------------
    addi    sp, sp, -64
    sw      ra,  60(sp)
    sw      s0,  56(sp)
    fsw     fs0, 52(sp)
    fsw     fs1, 48(sp)
    fsw     fs2, 44(sp)
    fsw     fs3, 40(sp)
    fsw     fs4, 36(sp)
    fsw     fs5, 32(sp)
    fsw     fs6, 28(sp)
    fsw     fs7, 24(sp)
    fsw     fs8, 20(sp)
    fsw     fs9, 16(sp)
    fsw     fs10, 12(sp)
    fsw     fs11, 8(sp)

    # Сохраняем указатели
    mv      s0, a2              # s0 = output pointer

    # ===================================================================
    # Загрузка входных атрибутов
    # ===================================================================

    # in_position = %1 (vec4): x,y,z,w
    flw     fs0,  0(a0)         # fs0 = pos.x  (%1.x)
    flw     fs1,  4(a0)         # fs1 = pos.y  (%1.y)
    flw     fs2,  8(a0)         # fs2 = pos.z  (%1.z)
    flw     fs3, 12(a0)         # fs3 = pos.w  (%1.w)

    # in_color = %5 (vec4): x,y,z,w
    flw     fs4, 16(a0)         # fs4 = color.x (%5.x)
    flw     fs5, 20(a0)         # fs5 = color.y (%5.y)
    flw     fs6, 24(a0)         # fs6 = color.z (%5.z)
    # color.w не используется

    # in_normal = %3 (vec3): x,y,z
    flw     fs7, 32(a0)         # fs7 = normal.x (%3.x)
    flw     fs8, 36(a0)         # fs8 = normal.y (%3.y)
    flw     fs9, 40(a0)         # fs9 = normal.z (%3.z)

    # ===================================================================
    # ЧАСТЬ 1: gl_Position = mvpMatrix * in_position
    # mvpMatrix rows: %11(0x40), %13(0x50), %15(0x60), %17(0x70)
    #
    # Вычисление:
    #   result = col3*w + col2*z + col1*y + col0*x
    #   (матрица хранится построчно, но шейдер умножает как column-major)
    #
    # %18..%21 = %17 * pos.w
    # %22..%25 = %15 * pos.z
    # %26..%29 = add
    # %30..%33 = %13 * pos.y
    # %34..%37 = add
    # %38..%41 = %11 * pos.x
    # %42..%45 = add  → gl_Position
    # ===================================================================

    # --- %17 * pos.w ---
    flw     ft0, 0x70(a1)       # %17.x
    flw     ft1, 0x74(a1)       # %17.y
    flw     ft2, 0x78(a1)       # %17.z
    flw     ft3, 0x7C(a1)       # %17.w
    fmul.s  ft0, ft0, fs3       # %18 = %17.x * pos.w
    fmul.s  ft1, ft1, fs3       # %19 = %17.y * pos.w
    fmul.s  ft2, ft2, fs3       # %20 = %17.z * pos.w
    fmul.s  ft3, ft3, fs3       # %21 = %17.w * pos.w

    # --- %15 * pos.z + previous ---
    flw     ft4, 0x60(a1)       # %15.x
    flw     ft5, 0x64(a1)       # %15.y
    flw     ft6, 0x68(a1)       # %15.z
    flw     ft7, 0x6C(a1)       # %15.w
    fmul.s  ft4, ft4, fs2       # %22 = %15.x * pos.z
    fmul.s  ft5, ft5, fs2       # %23 = %15.y * pos.z
    fmul.s  ft6, ft6, fs2       # %24 = %15.z * pos.z
    fmul.s  ft7, ft7, fs2       # %25 = %15.w * pos.z
    fadd.s  ft0, ft4, ft0       # %26 = %22 + %18
    fadd.s  ft1, ft5, ft1       # %27 = %23 + %19
    fadd.s  ft2, ft6, ft2       # %28 = %24 + %20
    fadd.s  ft3, ft7, ft3       # %29 = %25 + %21

    # --- %13 * pos.y + previous ---
    flw     ft4, 0x50(a1)       # %13.x
    flw     ft5, 0x54(a1)       # %13.y
    flw     ft6, 0x58(a1)       # %13.z
    flw     ft7, 0x5C(a1)       # %13.w
    fmul.s  ft4, ft4, fs1       # %30 = %13.x * pos.y
    fmul.s  ft5, ft5, fs1       # %31 = %13.y * pos.y
    fmul.s  ft6, ft6, fs1       # %32 = %13.z * pos.y
    fmul.s  ft7, ft7, fs1       # %33 = %13.w * pos.y
    fadd.s  ft0, ft4, ft0       # %34 = %30 + %26
    fadd.s  ft1, ft5, ft1       # %35 = %31 + %27
    fadd.s  ft2, ft6, ft2       # %36 = %32 + %28
    fadd.s  ft3, ft7, ft3       # %37 = %33 + %29

    # --- %11 * pos.x + previous ---
    flw     ft4, 0x40(a1)       # %11.x
    flw     ft5, 0x44(a1)       # %11.y
    flw     ft6, 0x48(a1)       # %11.z
    flw     ft7, 0x4C(a1)       # %11.w
    fmul.s  ft4, ft4, fs0       # %38 = %11.x * pos.x
    fmul.s  ft5, ft5, fs0       # %39 = %11.y * pos.x
    fmul.s  ft6, ft6, fs0       # %40 = %11.z * pos.x
    fmul.s  ft7, ft7, fs0       # %41 = %11.w * pos.x
    fadd.s  ft0, ft4, ft0       # %42 = gl_Position.x
    fadd.s  ft1, ft5, ft1       # %43 = gl_Position.y
    fadd.s  ft2, ft6, ft2       # %44 = gl_Position.z
    fadd.s  ft3, ft7, ft3       # %45 = gl_Position.w

    # Сохраняем gl_Position в выходной буфер
    fsw     ft0, 0(s0)          # gl_Position.x
    fsw     ft1, 4(s0)          # gl_Position.y
    fsw     ft2, 8(s0)          # gl_Position.z
    fsw     ft3, 12(s0)         # gl_Position.w

    # ===================================================================
    # ЧАСТЬ 2: eyePos = mvMatrix * in_position
    # mvMatrix rows: %67(0x00), %69(0x10), %71(0x20), %73(0x30)
    #
    # %74..%77 = %73 * pos.w
    # %78..%81 = %71 * pos.z
    # ... и так далее до %98..%101
    # Затем: perspective divide: %102=%98/%101, %103=%99/%101, %104=%100/%101
    # ===================================================================

    # --- %73 * pos.w ---
    flw     ft0, 0x30(a1)       # %73.x
    flw     ft1, 0x34(a1)       # %73.y
    flw     ft2, 0x38(a1)       # %73.z
    flw     ft3, 0x3C(a1)       # %73.w
    fmul.s  ft0, ft0, fs3       # %74
    fmul.s  ft1, ft1, fs3       # %75
    fmul.s  ft2, ft2, fs3       # %76
    fmul.s  ft3, ft3, fs3       # %77

    # --- %71 * pos.z + previous ---
    flw     ft4, 0x20(a1)       # %71.x
    flw     ft5, 0x24(a1)       # %71.y
    flw     ft6, 0x28(a1)       # %71.z
    flw     ft7, 0x2C(a1)       # %71.w
    fmul.s  ft4, ft4, fs2       # %78
    fmul.s  ft5, ft5, fs2       # %79
    fmul.s  ft6, ft6, fs2       # %80
    fmul.s  ft7, ft7, fs2       # %81
    fadd.s  ft0, ft4, ft0       # %82
    fadd.s  ft1, ft5, ft1       # %83
    fadd.s  ft2, ft6, ft2       # %84
    fadd.s  ft3, ft7, ft3       # %85

    # --- %69 * pos.y + previous ---
    flw     ft4, 0x10(a1)       # %69.x
    flw     ft5, 0x14(a1)       # %69.y
    flw     ft6, 0x18(a1)       # %69.z
    flw     ft7, 0x1C(a1)       # %69.w
    fmul.s  ft4, ft4, fs1       # %86
    fmul.s  ft5, ft5, fs1       # %87
    fmul.s  ft6, ft6, fs1       # %88
    fmul.s  ft7, ft7, fs1       # %89
    fadd.s  ft0, ft4, ft0       # %90
    fadd.s  ft1, ft5, ft1       # %91
    fadd.s  ft2, ft6, ft2       # %92
    fadd.s  ft3, ft7, ft3       # %93

    # --- %67 * pos.x + previous ---
    flw     ft4, 0x00(a1)       # %67.x
    flw     ft5, 0x04(a1)       # %67.y
    flw     ft6, 0x08(a1)       # %67.z
    flw     ft7, 0x0C(a1)       # %67.w
    fmul.s  ft4, ft4, fs0       # %94
    fmul.s  ft5, ft5, fs0       # %95
    fmul.s  ft6, ft6, fs0       # %96
    fmul.s  ft7, ft7, fs0       # %97
    fadd.s  ft0, ft4, ft0       # %98 = eyePos.x
    fadd.s  ft1, ft5, ft1       # %99 = eyePos.y
    fadd.s  ft2, ft6, ft2       # %100 = eyePos.z
    fadd.s  ft3, ft7, ft3       # %101 = eyePos.w

    # --- Perspective divide ---
    # %102 = %98 / %101
    # %103 = %99 / %101
    # %104 = %100 / %101
    fdiv.s  ft0, ft0, ft3       # %102 = eyePos.x / eyePos.w
    fdiv.s  ft1, ft1, ft3       # %103 = eyePos.y / eyePos.w
    fdiv.s  ft2, ft2, ft3       # %104 = eyePos.z / eyePos.w

    # --- Negate: %105 = -eyePos.x/w, %106 = -eyePos.y/w, %107 = -eyePos.z/w ---
    fsgnjn.s ft0, ft0, ft0     # %105 = -(%102)
    fsgnjn.s ft1, ft1, ft1     # %106 = -(%103)
    fsgnjn.s ft2, ft2, ft2     # %107 = -(%104)

    # ===================================================================
    # ЧАСТЬ 3: Вычисление направления света (lightDir) и нормализация
    #
    # lightPos = (2.0, 2.0, 20.0)  — константа %6
    # vLightDir = lightPos - eyePos_divided  (= lightPos + negated)
    #   %108 = 2.0 + %105
    #   %109 = 2.0 + %106
    #   %110 = 20.0 + %107
    #
    # length = sqrt(%108^2 + %109^2 + %110^2)
    # normalized = lightDir / length
    # ===================================================================

    # Загружаем константы для позиции света
    lui     t0, %hi(.Lconst_2_0)
    flw     ft3, %lo(.Lconst_2_0)(t0)      # ft3 = 2.0
    lui     t0, %hi(.Lconst_20_0)
    flw     ft4, %lo(.Lconst_20_0)(t0)     # ft4 = 20.0

    fadd.s  ft0, ft3, ft0      # %108 = 2.0 + (-eyeX/w)
    fadd.s  ft1, ft3, ft1      # %109 = 2.0 + (-eyeY/w)
    fadd.s  ft2, ft4, ft2      # %110 = 20.0 + (-eyeZ/w)

    # lengthSq = %110^2 + %109^2 + %108^2
    fmul.s  ft3, ft2, ft2      # %111 = %110 * %110
    fmul.s  ft4, ft1, ft1      # %112 = %109 * %109
    fadd.s  ft3, ft3, ft4      # %113 = %111 + %112
    fmul.s  ft4, ft0, ft0      # %114 = %108 * %108
    fadd.s  ft3, ft3, ft4      # %115 = %113 + %114 (= lengthSq)

    # length = sqrt(lengthSq)
    fsqrt.s ft3, ft3            # %116 = sqrt(%115)

    # normalize
    fdiv.s  ft0, ft0, ft3      # %117 = lightDir.x normalized
    fdiv.s  ft1, ft1, ft3      # %118 = lightDir.y normalized
    fdiv.s  ft2, ft2, ft3      # %119 = lightDir.z normalized

    # ===================================================================
    # ЧАСТЬ 4: Трансформация нормали (normalMatrix * in_normal)
    # normalMatrix rows: %47(0x80), %49(0x90), %51(0xA0)
    #
    # %52..%54 = %51 * normal.z
    # %55..%57 = %49 * normal.y
    # %58..%60 = add
    # %61..%63 = %47 * normal.x
    # %64..%66 = add → transformed normal
    # ===================================================================

    # --- %51 * normal.z ---
    flw     ft4, 0xA0(a1)      # %51.x
    flw     ft5, 0xA4(a1)      # %51.y
    flw     ft6, 0xA8(a1)      # %51.z
    fmul.s  ft4, ft4, fs9      # %52 = %51.x * normal.z
    fmul.s  ft5, ft5, fs9      # %53 = %51.y * normal.z
    fmul.s  ft6, ft6, fs9      # %54 = %51.z * normal.z

    # --- %49 * normal.y + previous ---
    flw     ft7, 0x90(a1)      # %49.x
    flw     ft8, 0x94(a1)      # %49.y
    flw     ft9, 0x98(a1)      # %49.z
    fmul.s  ft7, ft7, fs8      # %55 = %49.x * normal.y
    fmul.s  ft8, ft8, fs8      # %56 = %49.y * normal.y
    fmul.s  ft9, ft9, fs8      # %57 = %49.z * normal.y
    fadd.s  ft4, ft7, ft4      # %58 = %55 + %52
    fadd.s  ft5, ft8, ft5      # %59 = %56 + %53
    fadd.s  ft6, ft9, ft6      # %60 = %57 + %54

    # --- %47 * normal.x + previous ---
    flw     ft7, 0x80(a1)      # %47.x
    flw     ft8, 0x84(a1)      # %47.y
    flw     ft9, 0x88(a1)      # %47.z
    fmul.s  ft7, ft7, fs7      # %61 = %47.x * normal.x
    fmul.s  ft8, ft8, fs7      # %62 = %47.y * normal.x
    fmul.s  ft9, ft9, fs7      # %63 = %47.z * normal.x
    fadd.s  ft4, ft7, ft4      # %64 = transformedNormal.x
    fadd.s  ft5, ft8, ft5      # %65 = transformedNormal.y
    fadd.s  ft6, ft9, ft6      # %66 = transformedNormal.z

    # ===================================================================
    # ЧАСТЬ 5: Dot product (освещение) и модуляция цвета
    #
    # dot = normal.z * lightDir.z + normal.y * lightDir.y + normal.x * lightDir.x
    # %120 = %66 * %119
    # %121 = %65 * %118
    # %122 = %120 + %121
    # %123 = %64 * %117
    # %124 = %122 + %123
    # %125 = max(0.0, %124)
    #
    # vVaryingColor = (color.x * %125, color.y * %125, color.z * %125, 1.0)
    # ===================================================================

    fmul.s  ft7, ft6, ft2      # %120 = transNorm.z * lightDir.z
    fmul.s  ft8, ft5, ft1      # %121 = transNorm.y * lightDir.y
    fadd.s  ft7, ft7, ft8      # %122 = %120 + %121
    fmul.s  ft8, ft4, ft0      # %123 = transNorm.x * lightDir.x
    fadd.s  ft7, ft7, ft8      # %124 = dot product

    # max(0.0, dot)
    # Загружаем 0.0
    fcvt.s.w ft8, zero          # ft8 = 0.0  (конвертируем целый 0 → float 0.0)
    fmax.s  ft7, ft8, ft7      # %125 = max(0.0, dot)

    # Модуляция цвета
    fmul.s  ft0, fs4, ft7      # %126 = color.x * intensity
    fmul.s  ft1, fs5, ft7      # %127 = color.y * intensity
    fmul.s  ft2, fs6, ft7      # %128 = color.z * intensity

    # Загружаем 1.0 для alpha
    lui     t0, %hi(.Lconst_1_0)
    flw     ft3, %lo(.Lconst_1_0)(t0)     # ft3 = 1.0

    # Сохраняем vVaryingColor в выходной буфер
    fsw     ft0, 16(s0)         # vVaryingColor.x
    fsw     ft1, 20(s0)         # vVaryingColor.y
    fsw     ft2, 24(s0)         # vVaryingColor.z
    fsw     ft3, 28(s0)         # vVaryingColor.w

    # ===================================================================
    # Эпилог: восстанавливаем регистры и выходим
    # ===================================================================
    flw     fs11, 8(sp)
    flw     fs10, 12(sp)
    flw     fs9, 16(sp)
    flw     fs8, 20(sp)
    flw     fs7, 24(sp)
    flw     fs6, 28(sp)
    flw     fs5, 32(sp)
    flw     fs4, 36(sp)
    flw     fs3, 40(sp)
    flw     fs2, 44(sp)
    flw     fs1, 48(sp)
    flw     fs0, 52(sp)
    lw      s0,  56(sp)
    lw      ra,  60(sp)
    addi    sp, sp, 64
    ret

# ===================================================================
# Секция данных (read-only константы)
# ===================================================================
    .section .rodata
    .align  2

.Lconst_1_0:
    .word   0x3F800000          # 1.0f

.Lconst_2_0:
    .word   0x40000000          # 2.0f

.Lconst_20_0:
    .word   0x41A00000          # 20.0f

    .size   vertex_shader_main, .-vertex_shader_main
